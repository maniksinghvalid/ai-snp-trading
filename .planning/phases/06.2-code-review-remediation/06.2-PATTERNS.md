```python
def insert_trade(self, trade: dict) -> None:
    """Insert a completed trade row into the trades table (2.3 fix).

    Must be called from _on_exit_fill when remaining_quantity reaches 0.
    trade dict keys: trade_id, position_id, code, entry_price, exit_price,
                     quantity, exit_reason, r_multiple, closed_at.
    """
    with self._lock:
        self._conn.execute(
            """INSERT OR IGNORE INTO trades
               (trade_id, position_id, code, entry_price, exit_price,
                quantity, exit_reason, r_multiple, closed_at)
               VALUES (?,?,?,?,?,?,?,?,?)""",
            (
                trade["trade_id"],
                trade["position_id"],
                trade["code"],
                trade["entry_price"],
                trade["exit_price"],
                trade["quantity"],
                trade["exit_reason"],
                trade.get("r_multiple"),
                trade["closed_at"],
            ),
        )
        self._conn.commit()
```
